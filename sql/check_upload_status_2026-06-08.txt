-- Q1. The last ~15 import batches across all sheets, newest first.
--     Look for: a DBVentas_ConMG row (did ventas create a batch at all?),
--     its status (success / partial_success / failed / processing), and any
--     'failed' rows from the mis-clicked upload + its notes (the real error).
SELECT
  source_sheet_name,
  status,
  rows_detected,
  rows_loaded,
  rows_failed,
  left(id::text, 8) AS batch,
  imported_at,
  updated_at,
  notes
FROM ingest.import_batches
ORDER BY COALESCE(updated_at, imported_at) DESC
LIMIT 15;
